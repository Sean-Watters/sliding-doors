\begin{frame}{Logic as a Specification Language (1)}
  A \textbf{logic} is a formal system for making statements of fact.
  eg: propositional logic, first-order logic, etc.

  \bigskip{}
  \pause

  Traditionally, we define the meaning of formulae relative to some mathematical structure called a \textbf{model}.
  (Model-theoretic semantics).

  \bigskip{}
  \pause

  The \textbf{model checking problem} for a logic is the problem of deciding whether, given a formula and model of the logic, the formula is validated in that model.

  \bigskip{}
  \pause

  Model Checking as a \textbf{formal verification} technique involves representing the system undergoing verification as a model, and the properties being verified as formulas.
  As long as the model checking problem for the logic is decidable, we can algorithmically verify that the property is true.
\end{frame}

\begin{frame}{Logic as a Specification Language (2)}

  Two questions:
  \bigskip{}
  \begin{enumerate}
    \item What is the right notion of model for real-world systems?
    \smallskip
    \item Which logics (best) let us reason about such models?
  \end{enumerate}

\end{frame}

\begin{frame}{Propositional Logic}
  Given some set $A$ of propositional atoms, for all $a \in \mathsf{At}$, let PL be the set of terms generated by:
  \begin{equation*}
    \varphi := a~|~\neg \varphi~|~\varphi \land \varphi~|~\varphi \lor \varphi~|~\varphi \Rightarrow \varphi
  \end{equation*}

  Models are valuation functions $\mathsf{At} \to 2$.

  \bigskip{}
  \pause

  \begin{block}{Propositional Logic for Model Checking?}
  \begin{itemize}
          \smallskip
    \item \textbf{The Good}: Model checking problem is decidable.
          \smallskip{}
    \item \textbf{The Bad}: Not very expressive; purely propositional encodings are clunky.
  \end{itemize}
  \end{block}

  \bigskip

  How about first-order logic? Much more expressive, but too expressive! Not decidable!
\end{frame}

\begin{frame}{Kripke Semantics (1)}

  \textbf{Key idea}: Computers and programs tend to progress in discreet time steps, with their properties potentially changing at each step.

  \bigskip

  \center Let's consider models that reflect that!

\end{frame}

\begin{frame}{Kripke Semantics (2)}

\begin{definition}
  A \textbf{Kripke model} is a tuple $(S,T,V)$ of a set of states $S$, a transition relation $T : S \to S \to \mathsf{Prop}$, and a valuation function $V : A \to S \to \mathsf{Prop}$ (given some set $A$ of atomic propositions).
\end{definition}

\bigskip

Notice that:
\begin{itemize}
  \item $S$ and $T$ induce a (possibly infinite) graph-like structure.
        \smallskip
  \item Each state $s \in S$ is equipped with a model of propositional logic, $a \mapsto V~a~s$.
        \smallskip
  \item Partial application of $T$ on some state $s \in S$ yields the ``is a successor of $s$'' predicate.
\end{itemize}

\end{frame}

\begin{frame}{Modal Logic (1)}
  For all propositional atoms $a \in \mathsf{At}$, let ML be the set of terms generated by:
  \begin{equation*}
    \varphi := a~|~\neg \varphi~|~\varphi \land \varphi~|~\varphi \lor \varphi~|~\varphi \Rightarrow \varphi~|~\Box \varphi~|~\Diamond \varphi
  \end{equation*}

  \bigskip

  Given a Kripke model $\mathcal{M} = (S,T,V)$, let $\meaningof{-}~- : ML \to S \to \mathsf{Prop}$, where:
  \begin{align*}
    \meaningof{a}~s &:= V~a~s \\
    \meaningof{\lnot \varphi}~s &:= \lnot~(\meaningof{\varphi}~s) \\
    \meaningof{\varphi \land \psi}~s &:= (\meaningof{\varphi}~s) \times (\meaningof{\psi}~s) \\
    \meaningof{\varphi \lor \psi}~s &:= (\meaningof{\varphi}~s) + (\meaningof{\psi}~s) \\
    \meaningof{\varphi \Rightarrow \psi}~s &:= (\meaningof{\varphi}~s) \to (\meaningof{\psi}~s) \\
    \meaningof{\Box \varphi}~s &:= \forall t \in (T~s).~\meaningof{\varphi}~t \\
    \meaningof{\Diamond \varphi}~s &:= \exists t \in (T~s).~\meaningof{\varphi}~t
  \end{align*}
\end{frame}

\begin{frame}{Modal Logic (2)}

\begin{block}{ML for Model Checking?}
\begin{itemize}
  \item \textbf{Good}: Model checking problem is still decidable.
        \smallskip{}
  \item \textbf{Good}: Kripke models are well-suited to representing real systems.
        \smallskip{}
  \item \textbf{Bad}: Expressivity is still limited; can only reason about fixed, finite time steps.
\end{itemize}
\end{block}

\bigskip
\pause

We cannot express, for example:

\begin{itemize}
  \item $\varphi$ is always true.
  \item $\varphi$ eventually becomes true, in at least one possible future.
  \item $\varphi$ remains true at least until $\psi$ becomes true, in all possible futures.
\end{itemize}

\center \textbf{We currently can't reason about infinte or unbounded behaviour.}

\center There's a neat way around that\ldots

\begin{frame}{The Modal $\mu$-Calculus (1)}
  For all propositional atoms $a \in \mathsf{At}$ and variable names $x \in \mathsf{Var}$, let $\mu$ML be the set of terms generated by:
  \begin{equation*}
    \varphi := a~|~\neg a~|~x~|~\varphi \land \varphi~|~\varphi \lor \varphi~|~\Box \varphi~|~\Diamond \varphi~|~\mu x. \varphi~|~\nu x. \varphi
  \end{equation*}

  \bigskip
  \pause

  Given a Kripke model $\mathcal{M}$ and an intepretation of free variables $i : \mathsf{Var} \to S \to \mathsf{Prop}$, let:
  \begin{align*}
    \meaningof{x}_{i}~s &:= i~x~s \\
    \meaningof{\mu x.~\varphi}_{i}~s &:= \mu~((U : S \to \mathsf{Prop}) \mapsto \meaningof{\varphi}_{i[x:=U]})~s \\
    \meaningof{\nu x.~\varphi}_{i}~s &:= \nu~((U : S \to \mathsf{Prop}) \mapsto \meaningof{\varphi}_{i[x:=U]})~s
  \end{align*}

  Where $i[x:=U]$ denotes updating the mapping of $x$ to $U$ in $i$.
  % subsumes all of the above.
  % note: strict positivity
  % mention semantics; classical, game, container
\end{frame}

\begin{frame}{The Modal $\mu$-Calculus (2)}
  Syntax:
  \begin{equation*}
    \varphi := a~|~\neg a~|~x~|~\varphi \land \varphi~|~\varphi \lor \varphi~|~\Box \varphi~|~\Diamond \varphi~|~\mu x. \varphi~|~\nu x. \varphi
  \end{equation*}

  \bigskip

  Semantics of $\mu x.~\varphi$:
  \begin{equation*}
    \meaningof{\mu x.~\varphi}_{i} := \mu~(U \mapsto \meaningof{\varphi}_{i[x:=U]})
  \end{equation*}

  \bigskip

  \textbf{Important notes}:
          \smallskip
  \begin{itemize}
    \item Not every map has well-defined least/greatest fixpoints. Kleene's Fixpoint Theorem guarantees that this one will, so long as it is monotone.
          \smallskip
    \item $\meaningof{-}$ being strictly positive guarantees that this will be the case.
          \smallskip
    \item This is why the syntax of $\mu$ML does not allow arbitrary negations or implication.
  \end{itemize}

\end{frame}

\begin{frame}{Fixpoint Unfolding}
  At the heart of the $\mu$-calculus is the semantic equivalance:
    \begin{equation*}
      \eta x.~\varphi~\equiv~\varphi [ \eta x.\varphi ~/~x]
    \end{equation*}

  We call $\varphi [ \eta x.\varphi~/~x ]$ the \textbf{unfolding} of $\eta x.~\varphi$.

\bigskip

  For example: let $E(p) := \mu x.~p \lor \Diamond x$. Then:
  \begin{align*}
    &E(p) \\
    \equiv~&p \lor \Diamond(E(p)) \\
    \equiv~&p \lor \Diamond(p \lor \Diamond(E(p))) \\
    \equiv~&\ldots
  \end{align*}

  \bigskip

  This being a \emph{least} fixpoint says that the formula will be validated in \emph{finitely} many unfoldings.

  % substitution (hand wavy, will do properly with dB in part 2)

\end{frame}

\begin{frame}{The Closure (1)}
  \begin{definition}
    The \textbf{closure} of a formula $\varphi$ is the minimal set which contains $\varphi$, and is closed under taking unfoldings of fixpoint formulas, and direct subformulas of non-fixpoint formulas.
  \end{definition}

  \bigskip

  In other words, it is the minimal set $C$ satisfying:
    \begin{align*}
      &\varphi \in C \\
      \bigcirc \varphi \in C \Rightarrow~&\varphi \in C,~\mathsf{where}~\bigcirc \in \{\Box,\Diamond\} \\
      \varphi \star \psi \in C \Rightarrow~&\varphi \in C~\mathsf{and}~\psi \in C,~\mathsf{where}~\star \in \{\land,\lor\} \\
      \eta x. \varphi \in C \Rightarrow~&\varphi [ \mu x. \varphi~/~x] \in C,~\mathsf{where}~\eta \in \{\mu,\nu\}
    \end{align*}
  % insability under alpha equivalence
  % relationship to model checking
\end{frame}

\begin{frame}{The Closure (2)}
  The closure encapsulates the semantics of a formula in a syntactic way:

  \bigskip

  Its graph directly yields an automaton that accepts (possibly infinite) paths through Kripke models that validate the formula.

  \bigskip

  Some facts about the closure:
          \smallskip

  \begin{enumerate}
    \item It is always non-empty (trivial).
          \smallskip
          \pause
    \item It is always finite (a bit less trivial).
          \smallskip
          \pause
    \item Fixpoint formulae and their unfoldings always have the same closure.
          \smallskip
          \pause
    \item If $\varphi$ and $\psi$ have the same closure, then $\varphi \equiv \psi$.
          \smallskip
          \pause
    \item But the other direction does not hold; semantically equivalent formulae may have different closures.
          \smallskip
          \pause
    \item Worse: even if $\varphi =_{\alpha} \psi$, then they may still have different closures (up to $\alpha$-equivalence). (Wickedness! Heresy!)
  \end{enumerate}



\end{frame}

\begin{frame}{Finiteness of the Closure (1)}
  \begin{theorem}
    For all $\varphi$, the closure of $\varphi$ is finite.
  \end{theorem}
  \bigskip
  \textbf{Intuition}: Unfoldings can increase the size of the formula, but the number of new subformulas always decreases.

  \vspace{10em}

\end{frame}

\begin{frame}{Closures of Unfoldings}
  \begin{theorem}
    \item For any fixpoint formula $\eta x. \varphi$, both it and its unfolding $\varphi [ \eta x. \varphi~/~x ]$ have the same closure.
  \end{theorem}

  \vspace{14em}

\end{frame}

\begin{frame}{Instability Under Alpha Equivalence}
  \begin{theorem}
    There exist formulas $\varphi$ and $\psi$ such that $\varphi =_{\alpha} \psi$, but $C_{\varphi} \neq C_{\psi}$.
  \end{theorem}
  \smallskip
  \center\emph{``That's just not nice!''} - Sam Lindley, 2024

  \vspace{12em}

\end{frame}
